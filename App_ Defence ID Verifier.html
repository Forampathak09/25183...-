"use client";

import { useState, useEffect } from "react";
import { Card, CardContent, CardDescription, CardFooter, CardHeader, CardTitle } from "@/components/ui/card";
import { Input } from "@/components/ui/input";
import { Label } from "@/components/ui/label";
import { Button } from "@/components/ui/button";
import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from "@/components/ui/select";
import { Textarea } from "@/components/ui/textarea";
import { Shield, Check, X, Upload, Clock, User, FileText, Camera, Eye, EyeOff, Lock, Unlock } from "lucide-react";
import { motion } from "framer-motion";

// Types
type VerificationStatus = "pending" | "in-progress" | "success" | "failed";
type DocumentType = "military-id" | "passport" | "driver-license" | "state-id" | "birth-certificate";
type RelationshipType = "spouse" | "child" | "parent" | "sibling" | "other";

interface VerificationRecord {
  id: string;
  date: string;
  status: "success" | "failed";
  name: string;
  relationship: string;
  accessPeriod: string;
}

export default function AdvancedNationalIDPortal() {
  // Form state
  const [step, setStep] = useState(1);
  const [formData, setFormData] = useState({
    name: "",
    idNumber: "",
    relationship: "" as RelationshipType,
    documentType: "" as DocumentType,
    documentNumber: "",
    additionalInfo: "",
    biometricConsent: false
  });
  
  // UI state
  const [isVerifying, setIsVerifying] = useState(false);
  const [verificationStatus, setVerificationStatus] = useState<VerificationStatus>("pending");
  const [verificationResult, setVerificationResult] = useState<string | null>(null);
  const [uploadedFiles, setUploadedFiles] = useState<File[]>([]);
  const [showBiometric, setShowBiometric] = useState(false);
  const [biometricProgress, setBiometricProgress] = useState(0);
  const [verificationHistory, setVerificationHistory] = useState<VerificationRecord[]>([]);
  const [showHistory, setShowHistory] = useState(false);
  
  // Initialize with sample history
  useEffect(() => {
    const sampleHistory: VerificationRecord[] = [
      {
        id: "VH001",
        date: "2023-10-15",
        status: "success",
        name: "John Smith",
        relationship: "spouse",
        accessPeriod: "30 days"
      },
      {
        id: "VH002",
        date: "2023-09-22",
        status: "success",
        name: "John Smith",
        relationship: "spouse",
        accessPeriod: "30 days"
      }
    ];
    setVerificationHistory(sampleHistory);
  }, []);

  const handleInputChange = (e: React.ChangeEvent<HTMLInputElement | HTMLTextAreaElement>) => {
    const { name, value } = e.target;
    setFormData(prev => ({ ...prev, [name]: value }));
  };

  const handleSelectChange = (name: string, value: string) => {
    setFormData(prev => ({ ...prev, [name]: value }));
  };

  const handleCheckboxChange = (name: string) => {
    setFormData(prev => ({ ...prev, [name]: !prev[name as keyof typeof formData] }));
  };

  const handleFileUpload = (e: React.ChangeEvent<HTMLInputElement>) => {
    if (e.target.files && e.target.files.length > 0) {
      const files = Array.from(e.target.files);
      setUploadedFiles(prev => [...prev, ...files]);
    }
  };

  const removeFile = (index: number) => {
    setUploadedFiles(prev => prev.filter((_, i) => i !== index));
  };

  const validateStep = (currentStep: number) => {
    if (currentStep === 1) {
      if (!formData.name.trim()) {
        setVerificationResult("Please enter your full name");
        return false;
      }
      
      if (!formData.idNumber.trim()) {
        setVerificationResult("Please enter your ID number");
        return false;
      }
      
      if (!formData.relationship) {
        setVerificationResult("Please select your relationship to defense personnel");
        return false;
      }
    }
    
    if (currentStep === 2) {
      if (!formData.documentType) {
        setVerificationResult("Please select a document type");
        return false;
      }
      
      if (!formData.documentNumber.trim()) {
        setVerificationResult("Please enter your document number");
        return false;
      }
      
      if (uploadedFiles.length === 0) {
        setVerificationResult("Please upload at least one supporting document");
        return false;
      }
    }
    
    return true;
  };

  const nextStep = () => {
    if (validateStep(step)) {
      setVerificationResult(null);
      setStep(prev => prev + 1);
    }
  };

  const prevStep = () => {
    setStep(prev => prev - 1);
    setVerificationResult(null);
  };

  const startBiometricVerification = () => {
    if (!formData.biometricConsent) {
      setVerificationResult("Please consent to biometric verification");
      return;
    }
    
    setShowBiometric(true);
    setVerificationStatus("in-progress");
    setBiometricProgress(0);
    
    // Simulate biometric verification progress
    const interval = setInterval(() => {
      setBiometricProgress(prev => {
        if (prev >= 100) {
          clearInterval(interval);
          completeVerification();
          return 100;
        }
        return prev + 10;
      });
    }, 300);
  };

  const completeVerification = () => {
    // Enhanced verification logic
    const isValid = formData.idNumber.length >= 6 && 
                   ["spouse", "child", "parent"].includes(formData.relationship) &&
                   formData.documentNumber.length >= 5 &&
                   uploadedFiles.length > 0;
    
    if (isValid) {
      setVerificationStatus("success");
      setVerificationResult("Identity verification successful! Your access has been granted for 30 days.");
      
      // Add to history
      const newRecord: VerificationRecord = {
        id: `VH${(verificationHistory.length + 100).toString().padStart(3, '0')}`,
        date: new Date().toISOString().split('T')[0],
        status: "success",
        name: formData.name,
        relationship: formData.relationship,
        accessPeriod: "30 days"
      };
      
      setVerificationHistory(prev => [newRecord, ...prev]);
    } else {
      setVerificationStatus("failed");
      setVerificationResult("Verification failed. Please ensure all documents are uploaded and details are correct.");
    }
    
    setIsVerifying(false);
  };

  const resetForm = () => {
    setStep(1);
    setFormData({
      name: "",
      idNumber: "",
      relationship: "" as RelationshipType,
      documentType: "" as DocumentType,
      documentNumber: "",
      additionalInfo: "",
      biometricConsent: false
    });
    setVerificationResult(null);
    setVerificationStatus("pending");
    setUploadedFiles([]);
    setShowBiometric(false);
    setBiometricProgress(0);
  };

  return (
    <div className="min-h-screen bg-gradient-to-b from-slate-900 to-slate-800 flex flex-col">
      {/* National Portal Header */}
      <header className="bg-white border-b border-slate-200 py-4 px-6">
        <div className="max-w-7xl mx-auto flex items-center justify-between">
          <div className="flex items-center space-x-3">
            <div className="bg-blue-700 p-2 rounded">
              <Shield className="h-8 w-8 text-white" />
            </div>
            <div>
              <h1 className="text-2xl font-bold text-slate-900">National Defense Portal</h1>
              <p className="text-sm text-slate-600">Official Government of [Country Name]</p>
            </div>
          </div>
          <div className="flex items-center space-x-4">
            <span className="text-sm text-slate-600">Secure Connection</span>
            <div className="h-3 w-3 rounded-full bg-green-500"></div>
          </div>
        </div>
      </header>

      {/* Main Content */}
      <main className="flex-grow flex items-center justify-center p-4">
        <div className="w-full max-w-4xl">
          <Card className="shadow-2xl border-0 bg-white/90 backdrop-blur-sm">
            <CardHeader className="text-center pb-6">
              <div className="flex justify-center mb-4">
                <div className="bg-blue-100 p-4 rounded-full">
                  <Shield className="h-10 w-10 text-blue-700" />
                </div>
              </div>
              <CardTitle className="text-3xl font-bold text-slate-900">Advanced ID Verification Portal</CardTitle>
              <CardDescription className="text-lg text-slate-700 mt-2">
                Official multi-factor verification system for defense personnel and families
              </CardDescription>
              
              {/* Progress Indicator */}
              <div className="mt-6">
                <div className="flex justify-center">
                  {[1, 2, 3].map((s) => (
                    <div key={s} className="flex items-center">
                      <div className={`flex items-center justify-center w-10 h-10 rounded-full border-2 ${
                        step === s 
                          ? "border-blue-600 bg-blue-600 text-white" 
                          : step > s 
                            ? "border-green-500 bg-green-500 text-white" 
                            : "border-slate-300 bg-white text-slate-500"
                      }`}>
                        {step > s ? <Check className="h-5 w-5" /> : s}
                      </div>
                      {s < 3 && (
                        <div className={`w-20 h-1 ${step > s ? "bg-green-500" : "bg-slate-200"}`}></div>
                      )}
                    </div>
                  ))}
                </div>
                <div className="flex justify-between mt-3 text-sm text-slate-600">
                  <span>Personal Info</span>
                  <span>Document Verification</span>
                  <span>Biometric Check</span>
                </div>
              </div>
              
              <div className="mt-4 p-3 bg-blue-50 rounded-lg border border-blue-200">
                <p className="text-sm text-blue-800 font-medium flex items-center justify-center">
                  <Lock className="h-4 w-4 mr-2" />
                  This is a secure government system. All activities are monitored and recorded.
                </p>
              </div>
            </CardHeader>
            
            <CardContent>
              {/* Verification Status Banner */}
              {verificationResult && (
                <motion.div 
                  initial={{ opacity: 0, y: -10 }}
                  animate={{ opacity: 1, y: 0 }}
                  className={`p-5 rounded-lg mb-6 border ${
                    verificationStatus === "success" 
                      ? "bg-green-50 border-green-200" 
                      : verificationStatus === "failed"
                        ? "bg-red-50 border-red-200"
                        : "bg-blue-50 border-blue-200"
                  }`}
                >
                  <div className="flex items-start">
                    <div className={`flex-shrink-0 h-6 w-6 rounded-full flex items-center justify-center ${
                      verificationStatus === "success" 
                        ? "bg-green-500" 
                        : verificationStatus === "failed"
                          ? "bg-red-500"
                          : "bg-blue-500"
                    }`}>
                      {verificationStatus === "success" ? (
                        <Check className="h-4 w-4 text-white" />
                      ) : verificationStatus === "failed" ? (
                        <X className="h-4 w-4 text-white" />
                      ) : (
                        <Clock className="h-4 w-4 text-white" />
                      )}
                    </div>
                    <div className="ml-3">
                      <h3 className={`text-lg font-medium ${
                        verificationStatus === "success" 
                          ? "text-green-800" 
                          : verificationStatus === "failed"
                            ? "text-red-800"
                            : "text-blue-800"
                      }`}>
                        {verificationStatus === "success" 
                          ? "Verification Successful" 
                          : verificationStatus === "failed"
                            ? "Verification Failed"
                            : "Verification in Progress"}
                      </h3>
                      <p className={`mt-1 text-sm ${
                        verificationStatus === "success" 
                          ? "text-green-700" 
                          : verificationStatus === "failed"
                            ? "text-red-700"
                            : "text-blue-700"
                      }`}>
                        {verificationResult}
                      </p>
                    </div>
                  </div>
                </motion.div>
              )}
              
              <form className="space-y-6">
                {/* Step 1: Personal Information */}
                {step === 1 && (
                  <motion.div 
                    initial={{ opacity: 0 }}
                    animate={{ opacity: 1 }}
                    className="space-y-6"
                  >
                    <div className="space-y-2">
                      <Label htmlFor="name" className="text-slate-700 font-medium">Full Name</Label>
                      <Input
                        id="name"
                        name="name"
                        value={formData.name}
                        onChange={handleInputChange}
                        placeholder="Enter your full legal name"
                        className="py-6 text-base"
                      />
                    </div>
                    
                    <div className="space-y-2">
                      <Label htmlFor="idNumber" className="text-slate-700 font-medium">Government ID Number</Label>
                      <Input
                        id="idNumber"
                        name="idNumber"
                        value={formData.idNumber}
                        onChange={handleInputChange}
                        placeholder="Enter your official ID number"
                        className="py-6 text-base"
                      />
                    </div>
                    
                    <div className="space-y-2">
                      <Label htmlFor="relationship" className="text-slate-700 font-medium">Relationship to Defense Personnel</Label>
                      <Select 
                        value={formData.relationship} 
                        onValueChange={(value) => handleSelectChange("relationship", value)}
                      >
                        <SelectTrigger className="py-6 text-base">
                          <SelectValue placeholder="Select your relationship" />
                        </SelectTrigger>
                        <SelectContent>
                          <SelectItem value="spouse">Spouse</SelectItem>
                          <SelectItem value="child">Child</SelectItem>
                          <SelectItem value="parent">Parent</SelectItem>
                          <SelectItem value="sibling">Sibling</SelectItem>
                          <SelectItem value="other">Other Authorized Family Member</SelectItem>
                        </SelectContent>
                      </Select>
                    </div>
                  </motion.div>
                )}
                
                {/* Step 2: Document Verification */}
                {step === 2 && (
                  <motion.div 
                    initial={{ opacity: 0 }}
                    animate={{ opacity: 1 }}
                    className="space-y-6"
                  >
                    <div className="space-y-2">
                      <Label htmlFor="documentType" className="text-slate-700 font-medium">Document Type</Label>
                      <Select 
                        value={formData.documentType} 
                        onValueChange={(value) => handleSelectChange("documentType", value)}
                      >
                        <SelectTrigger className="py-6 text-base">
                          <SelectValue placeholder="Select document type" />
                        </SelectTrigger>
                        <SelectContent>
                          <SelectItem value="military-id">Military ID</SelectItem>
                          <SelectItem value="passport">Passport</SelectItem>
                          <SelectItem value="driver-license">Driver's License</SelectItem>
                          <SelectItem value="state-id">State ID Card</SelectItem>
                          <SelectItem value="birth-certificate">Birth Certificate</SelectItem>
                        </SelectContent>
                      </Select>
                    </div>
                    
                    <div className="space-y-2">
                      <Label htmlFor="documentNumber" className="text-slate-700 font-medium">Document Number</Label>
                      <Input
                        id="documentNumber"
                        name="documentNumber"
                        value={formData.documentNumber}
                        onChange={handleInputChange}
                        placeholder="Enter document number"
                        className="py-6 text-base"
                      />
                    </div>
                    
                    <div className="space-y-2">
                      <Label className="text-slate-700 font-medium">Upload Supporting Documents</Label>
                      <div className="border-2 border-dashed border-slate-300 rounded-lg p-6 text-center">
                        <Upload className="mx-auto h-12 w-12 text-slate-400" />
                        <div className="mt-4 flex text-sm text-slate-600 justify-center">
                          <label className="relative cursor-pointer bg-white rounded-md font-medium text-blue-600 hover:text-blue-500">
                            <span>Upload files</span>
                            <input 
                              type="file" 
                              className="sr-only" 
                              multiple 
                              onChange={handleFileUpload}
                            />
                          </label>
                          <p className="pl-1">or drag and drop</p>
                        </div>
                        <p className="text-xs text-slate-500 mt-2">
                          PNG, JPG, PDF up to 10MB
                        </p>
                      </div>
                      
                      {uploadedFiles.length > 0 && (
                        <div className="mt-4">
                          <h4 className="text-sm font-medium text-slate-700 mb-2">Uploaded Files:</h4>
                          <ul className="space-y-2">
                            {uploadedFiles.map((file, index) => (
                              <li key={index} className="flex items-center justify-between bg-slate-50 p-2 rounded">
                                <div className="flex items-center">
                                  <FileText className="h-4 w-4 text-slate-500 mr-2" />
                                  <span className="text-sm text-slate-700 truncate max-w-xs">{file.name}</span>
                                </div>
                                <Button 
                                  type="button" 
                                  variant="ghost" 
                                  size="sm" 
                                  onClick={() => removeFile(index)}
                                >
                                  <X className="h-4 w-4" />
                                </Button>
                              </li>
                            ))}
                          </ul>
                        </div>
                      )}
                    </div>
                  </motion.div>
                )}
                
                {/* Step 3: Biometric Verification */}
                {step === 3 && (
                  <motion.div 
                    initial={{ opacity: 0 }}
                    animate={{ opacity: 1 }}
                    className="space-y-6"
                  >
                    {!showBiometric ? (
                      <>
                        <div className="bg-blue-50 border border-blue-200 rounded-lg p-4">
                          <div className="flex">
                            <div className="flex-shrink-0">
                              <Shield className="h-5 w-5 text-blue-400" />
                            </div>
                            <div className="ml-3">
                              <h3 className="text-sm font-medium text-blue-800">Biometric Verification Required</h3>
                              <div className="mt-2 text-sm text-blue-700">
                                <p>For enhanced security, we require biometric verification. This process captures unique physical characteristics to confirm your identity.</p>
                              </div>
                            </div>
                          </div>
                        </div>
                        
                        <div className="flex items-center">
                          <input
                            id="biometric-consent"
                            type="checkbox"
                            checked={formData.biometricConsent}
                            onChange={() => handleCheckboxChange("biometricConsent")}
                            className="h-4 w-4 text-blue-600 border-slate-300 rounded"
                          />
                          <label htmlFor="biometric-consent" className="ml-2 block text-sm text-slate-700">
                            I consent to biometric data collection for identity verification purposes
                          </label>
                        </div>
                        
                        <div className="flex justify-center">
                          <Button 
                            type="button" 
                            onClick={startBiometricVerification}
                            className="px-8 py-6 text-lg font-semibold bg-blue-700 hover:bg-blue-800"
                            disabled={!formData.biometricConsent}
                          >
                            <Camera className="mr-2 h-5 w-5" />
                            Start Biometric Verification
                          </Button>
                        </div>
                      </>
                    ) : (
                      <div className="text-center py-8">
                        <div className="relative mx-auto w-64 h-64 rounded-full bg-slate-100 border-4 border-blue-500 flex items-center justify-center">
                          <div className="absolute inset-0 rounded-full border-4 border-blue-500 animate-ping opacity-20"></div>
                          <Camera className="h-16 w-16 text-blue-500" />
                        </div>
                        
                        <div className="mt-8">
                          <div className="w-full bg-slate-200 rounded-full h-2.5">
                            <div 
                              className="bg-blue-600 h-2.5 rounded-full transition-all duration-300" 
                              style={{ width: `${biometricProgress}%` }}
                            ></div>
                          </div>
                          <p className="mt-2 text-sm text-slate-600">
                            {biometricProgress < 100 
                              ? "Capturing biometric data..." 
                              : "Processing verification..."}
                          </p>
                        </div>
                      </div>
                    )}
                  </motion.div>
                )}
                
                {/* Navigation Buttons */}
                <div className="flex justify-between pt-6">
                  <Button 
                    type="button" 
                    variant="outline" 
                    onClick={prevStep}
                    disabled={step === 1 || showBiometric}
                    className="px-6"
                  >
                    Previous
                  </Button>
                  
                  {step < 3 ? (
                    <Button 
                      type="button" 
                      onClick={nextStep}
                      className="px-6 bg-blue-700 hover:bg-blue-800"
                    >
                      Next Step

                    <div className="flex space-x-3">
                      <Button 
                        type="button" 
                        variant="outline"
                        onClick={() => setShowHistory(!showHistory)}
                        className="px-6"
                      >
                        {showHistory ? "Hide History" : "Show History"}
                      </Button>
                      <Button 
                        type="button" 
                        onClick={resetForm}

                      >
                        Reset Form

              {/* Verification History */}
              {showHistory && (
                <motion.div 
                  initial={{ opacity: 0, height: 0 }}
                  animate={{ opacity: 1, height: "auto" }}
                  className="mt-8 border-t border-slate-200 pt-6"
                >
                  <h3 className="text-lg font-medium text-slate-900 mb-4">Verification History</h3>
                  <div className="space-y-3">
                    {verificationHistory.length > 0 ? (
                      verificationHistory.map((record) => (
                        <div 
                          key={record.id} 
                          className={`flex items-center justify-between p-4 rounded-lg border ${
                            record.status === "success" 
                              ? "bg-green-50 border-green-200" 
                              : "bg-red-50 border-red-200"
                          }`}
                        >
                          <div>
                            <div className="flex items-center">
                              <span className={`inline-flex items-center px-2.5 py-0.5 rounded-full text-xs font-medium ${
                                record.status === "success" 
                                  ? "bg-green-100 text-green-800" 
                                  : "bg-red-100 text-red-800"
                              }`}>
                                {record.status === "success" ? (
                                  <Check className="h-3 w-3 mr-1" />

                                  <X className="h-3 w-3 mr-1" />
                                )}
                                {record.status}
                              </span>
                              <span className="ml-3 text-sm text-slate-600">{record.date}</span>

                          </div>

                        </div>
                      ))
                    ) : (
                      <p className="text-center py-4 text-slate-500">No verification history found</p>
                    )}
                  </div>
                </motion.div>
              )}
            </CardContent>
            
            <CardFooter className="flex flex-col items-center pt-6 border-t border-slate-200">

              </div>
            </CardFooter>
          </Card>
          
          <div className="mt-6 text-center text-xs text-slate-400">

          </div>
        </div>
      </main>
    </div>
  );
